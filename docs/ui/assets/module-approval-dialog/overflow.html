import { slug } from "./slug";

export interface Piece { title: string; dependsOn: string[] }

export function planBatches(pieces: Piece[]): Piece[][] {
  const done = new Set<string>();
  const batches: Piece[][] = [];
  let left = [...pieces];
  while (left.length > 0) {
    const ready = left.filter((p) => p.dependsOn.every((d) => done.has(slug(d))));
    if (ready.length === 0) throw new Error("the pieces depend on each other in a circle");
    batches.push(ready);
    for (const p of ready) done.add(slug(p.title));
    left = left.filter((p) => !ready.includes(p));
  }
  return batches;
}

export function mergeResults(results: Record<string, string>[]): Record<string, string> {
  return Object.assign({}, ...results);
}

export function describe(batches: Piece[][]): string {
  return batches.map((b, i) => `${i + 1}: ${b.map((p) => p.title).join(", ")}`).join("\n");
}

export const slug = (title: string): string =>
  title
    .toLowerCase()
    .normalize("NFKD")
    .replace(/['"]/g, "")
    .replace(/[^a-z0-9]+/g, "-")
    .replace(/^-|-$/g, "");

export const unslug = (id: string): string =>
  id
    .split("-")
    .map((w) => w.charAt(0).toUpperCase() + w.slice(1))
    .join(" ");

export function uniqueSlug(title: string, taken: Set<string>): string {
  const base = slug(title);
  let candidate = base;
  let n = 2;
  while (taken.has(candidate)) candidate = `${base}-${n++}`;
  taken.add(candidate);
  return candidate;
}
